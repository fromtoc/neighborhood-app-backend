三、相關 API

3.1 註冊 FCM Device Token

PUT   /api/v1/notifications/token 需登入

App 啟動後取得 FCM token，呼叫此 API 註冊到後台。
後台以 userId + platform 為唯一鍵，自動新增或更新。

Request Body：

{ "token": "fMx7K9...(FCM device token)", "platform": "ios" // "ios" | "android" }

回傳：

{ "code": 200, "message": "success" }
DEL   /api/v1/notifications/token?platform=ios 需登入

登出時呼叫，移除 device token，停止推播。

3.2 通知收件匣

GET   /api/v1/notifications?page=1&size=20 需登入

取得通知列表（分頁），包含未讀數。

回傳範例：

{ "code": 200, "data": { "records": [ { "id": 123, "type": "new_info", "title": "【氣象特報】臺北市 大雨特報", "body": "臺北市大雨特報，請注意安全...", "refType": "post", "refId": 456, "isRead": 0, "createdAt": "2026-03-18T10:30:00", "neighborhoodId": 7742, "neighborhoodName": "堵南里", "city": "基隆市", "district": "七堵區" } ], "total": 48, "unread": 5, "page": 1, "size": 20 } }
PUT   /api/v1/notifications/{id}/read 需登入

標記單則通知為已讀。

PUT   /api/v1/notifications/read-all 需登入

一鍵全部已讀。

3.3 通知設定

GET   /api/v1/notifications/settings 需登入

取得使用者的通知偏好設定。

{ "code": 200, "data": { "userId": 1, "master": 1, // 總開關（0=全部關閉） "infoMedium": 1, // 中等資訊（空品/水情） "infoNormal": 1, // 一般資訊（新聞/公告） "garbageTruck": 1, // 垃圾車（預留） "newPost": 1, // 新貼文 "postReply": 1, // 貼文回覆 "chat": 1, // 聊天室 "privateMessage": 1 // 私訊 } }
PUT   /api/v1/notifications/settings 需登入

更新通知設定（只傳需要改的欄位）。

// 範例：關閉聊天通知 { "chat": 0 } // 範例：關閉總開關（所有通知停止） { "master": 0 }

四、通知點擊導頁規則

每則通知包含 refType + refId，App 根據這兩個欄位決定跳轉目的地：

refTyperefId導頁目的地
post貼文 ID貼文詳情頁
chat_message訊息 ID聊天室頁面
user對方的 userId私訊對話頁面
💡 neighborhoodId 的用途：通知帶有 neighborhoodId / city / district / neighborhoodName，可用於在通知列表顯示「來自哪個里」，或導頁時切換到對應的里。

5.2 Android（Kotlin）完整代碼

// ========== 1. build.gradle (app) ==========
// plugins { id 'com.google.gms.google-services' }
// dependencies { implementation 'com.google.firebase:firebase-messaging:24.0.0' }

// ========== 2. MyFirebaseMessagingService.kt ==========
class MyFirebaseMessagingService : FirebaseMessagingService() {

    // Token 更新時（首次安裝 / Token 刷新）
    override fun onNewToken(token: String) {
        super.onNewToken(token)
        // 儲存到本地，等登入後上傳
        PreferenceManager.getDefaultSharedPreferences(this)
            .edit().putString("fcm_token", token).apply()
        // 如果已登入，立即上傳
        uploadTokenIfLoggedIn(token)
    }

    // 前景收到推播
    override fun onMessageReceived(message: RemoteMessage) {
        val title = message.notification?.title ?: return
        val body  = message.notification?.body ?: ""

        // 顯示本地通知
        val channelId = "golocal_default"
        val intent = Intent(this, MainActivity::class.java)
        val pending = PendingIntent.getActivity(this, 0, intent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE)

        val notification = NotificationCompat.Builder(this, channelId)
            .setSmallIcon(R.drawable.ic_notification)
            .setContentTitle(title)
            .setContentText(body)
            .setAutoCancel(true)
            .setContentIntent(pending)
            .build()

        val manager = getSystemService(NotificationManager::class.java)
        // Android 8+ 需要 Channel
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            manager.createNotificationChannel(
                NotificationChannel(channelId, "GoLocal 通知",
                    NotificationManager.IMPORTANCE_DEFAULT))
        }
        manager.notify(System.currentTimeMillis().toInt(), notification)
    }
}

// ========== 3. AndroidManifest.xml ==========
// <service android:name=".MyFirebaseMessagingService"
//          android:exported="false">
//   <intent-filter>
//     <action android:name="com.google.firebase.MESSAGING_EVENT" />
//   </intent-filter>
// </service>

// ========== 4. 登入後上傳 Token ==========
fun registerFcmToken(authToken: String) {
    FirebaseMessaging.getInstance().token.addOnSuccessListener { fcmToken ->
        val body = mapOf("token" to fcmToken, "platform" to "android")
        api.put("/api/v1/notifications/token", body, authToken)
    }
}

// ========== 5. 登出時移除 Token ==========
fun unregisterFcmToken(authToken: String) {
    api.delete("/api/v1/notifications/token?platform=android", authToken)
}

5.3 iOS（Swift）完整代碼

// ========== 1. AppDelegate.swift ==========
import UIKit
import FirebaseCore
import FirebaseMessaging
import UserNotifications

@main
class AppDelegate: UIResponder, UIApplicationDelegate,
                    UNUserNotificationCenterDelegate,
                    MessagingDelegate {

    func application(_ application: UIApplication,
        didFinishLaunchingWithOptions launchOptions: [UIApplication.LaunchOptionsKey: Any]?) -> Bool {

        FirebaseApp.configure()

        // 設定通知代理
        UNUserNotificationCenter.current().delegate = self
        Messaging.messaging().delegate = self

        // 請求通知權限
        UNUserNotificationCenter.current().requestAuthorization(
            options: [.alert, .badge, .sound]) { granted, error in
            if granted {
                DispatchQueue.main.async { application.registerForRemoteNotifications() }
            }
        }
        return true
    }

    // APNs Token 取得後交給 Firebase
    func application(_ application: UIApplication,
        didRegisterForRemoteNotificationsWithDeviceToken deviceToken: Data) {
        Messaging.messaging().apnsToken = deviceToken
    }

    // FCM Token 更新
    func messaging(_ messaging: Messaging,
        didReceiveRegistrationToken fcmToken: String?) {
        guard let token = fcmToken else { return }
        UserDefaults.standard.set(token, forKey: "fcm_token")
        // 如果已登入，立即上傳
        uploadTokenIfLoggedIn(token)
    }

    // 前景收到推播 — 顯示 banner
    func userNotificationCenter(_ center: UNUserNotificationCenter,
        willPresent notification: UNNotification,
        withCompletionHandler handler: @escaping (UNNotificationPresentationOptions) -> Void) {
        handler([.banner, .badge, .sound])
    }

    // 使用者點擊通知
    func userNotificationCenter(_ center: UNUserNotificationCenter,
        didReceive response: UNNotificationResponse,
        withCompletionHandler handler: @escaping () -> Void) {
        let userInfo = response.notification.request.content.userInfo
        // 從 userInfo 解析 refType / refId 進行導頁
        handleNotificationTap(userInfo)
        handler()
    }
}

// ========== 2. 登入後上傳 Token ==========
func registerFcmToken(authToken: String) {
    Messaging.messaging().token { token, error in
        guard let token = token else { return }
        let body: [String: Any] = ["token": token, "platform": "ios"]
        api.put("/api/v1/notifications/token", body: body, token: authToken)
    }
}

// ========== 3. 登出時移除 Token ==========
func unregisterFcmToken(authToken: String) {
    api.delete("/api/v1/notifications/token?platform=ios", token: authToken)
}

七、推播 Payload 格式

後台發送的是 Notification Message（非 Data Message），格式如下：

{ "notification": { "title": "【氣象特報】臺北市 大雨特報", "body": "臺北市大雨特報，請注意安全..." } }
⚠️ 目前只傳 notification，不帶 data payload。
如果 App 需要在推播中攜帶 refType / refId 來實現點擊導頁，需要後台改為同時發送 data 欄位。屆時可以擴充。

如需攜帶 data（未來擴充）

後台只需在 MulticastMessage.builder() 加上 .putAllData()，App 端即可從 userInfo / remoteMessage.data 取得：

{ "notification": { "title": "...", "body": "..." }, "data": { "refType": "post", "refId": "456", "notifType": "new_info" } }
